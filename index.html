// ======================================================== // 🖼️ EXTERNAL IMAGE SPRITE ANIMATION LOGIC // ======================================================== // 1. Define paths for all 12 frame images const SPRITE_FILES = { down: ['down1.png', 'down2.png', 'down3.png'], up: ['up1.png', 'up2.png', 'up3.png'], left: ['left1.png', 'left2.png', 'left3.png'], right: ['right1.png', 'right2.png', 'right3.png'] }; // 2. Preload all image assets const playerSprites = { down: [], up: [], left: [], right: [] }; let spritesLoadedCount = 0; const totalSprites = 12; Object.keys(SPRITE_FILES).forEach(dir => { SPRITE_FILES[dir].forEach((path, index) => { const img = new Image(); img.src = path; img.onload = () => { spritesLoadedCount++; }; playerSprites[dir][index] = img; }); }); // Direction keys corresponding to sprite object keys const DIRECTION_KEYS = ['down', 'up', 'left', 'right']; // ======================================================== // 🚶 MOVEMENT & ANIMATION TIMER LOGIC // ======================================================== function updatePlayer(deltaTime) { if (player.isAttacking) { player.attackTimer -= deltaTime; if (player.attackTimer <= 0) player.isAttacking = false; } let dx = 0, dy = 0; if (keys['keyw'] || keys['arrowup']) dy -= 1; if (keys['keys'] || keys['arrowdown']) dy += 1; if (keys['keya'] || keys['arrowleft']) dx -= 1; if (keys['keyd'] || keys['arrowright']) dx += 1; if (joystick.active && dx === 0 && dy === 0) { const diffX = joystick.currX - joystick.startX; const diffY = joystick.currY - joystick.startY; const dist = Math.hypot(diffX, diffY); if (dist > 5) { dx = diffX / Math.max(dist, joystick.maxRadius); dy = diffY / Math.max(dist, joystick.maxRadius); } } player.isMoving = (dx !== 0 || dy !== 0); if (player.isMoving) { // Direction selection: 0 = down, 1 = up, 2 = left, 3 = right if (Math.abs(dx) > Math.abs(dy)) { player.direction = dx > 0 ? 3 : 2; } else { player.direction = dy > 0 ? 0 : 1; } // Cycle frame every 0.5 seconds (0 -> 1 -> 2 -> 0 maps to frame 1 -> 2 -> 3 -> 1) player.animTimer += deltaTime; if (player.animTimer >= 0.5) { player.animTimer = 0; player.animFrame = (player.animFrame + 1) % 3; } } else { // Reset to Frame 1 (Index 0) when idle player.animFrame = 0; player.animTimer = 0; } if (dx !== 0 && dy !== 0) { dx *= 0.7071; dy *= 0.7071; } let nextX = player.x + dx * player.speed * deltaTime; let nextY = player.y + dy * player.speed * deltaTime; const currentChunk = getOrCreateChunk(player.gridX, player.gridY); // Collisions for (const rock of currentChunk.rocks) { const dist = Math.hypot(nextX - rock.x, nextY - rock.y); const minDist = player.radius + rock.radius; if (dist < minDist) { const angle = Math.atan2(nextY - rock.y, nextX - rock.x); nextX = rock.x + Math.cos(angle) * minDist; nextY = rock.y + Math.sin(angle) * minDist; } } for (const tree of currentChunk.trees) { const dist = Math.hypot(nextX - tree.x, nextY - tree.y); const minDist = player.radius + tree.trunkRadius; if (dist < minDist) { const angle = Math.atan2(nextY - tree.y, nextX - tree.x); nextX = tree.x + Math.cos(angle) * minDist; nextY = tree.y + Math.sin(angle) * minDist; } } player.x = nextX; player.y = nextY; // Screen Chunk Transitions if (player.x > canvas.width) { player.gridX++; player.x = player.radius + 2; } else if (player.x < 0) { player.gridX--; player.x = canvas.width - player.radius - 2; } if (player.y < 0) { player.gridY++; player.y = canvas.height - player.radius - 2; } else if (player.y > canvas.height) { player.gridY--; player.y = player.radius + 2; } sectorEl.textContent = `X: ${player.gridX}, Y: ${player.gridY}`; } function drawPlayerSprite() { const dirKey = DIRECTION_KEYS[player.direction]; // 'down', 'up', 'left', or 'right' const spriteImg = playerSprites[dirKey][player.animFrame]; // Select frame 0, 1, or 2 if (spriteImg && spriteImg.complete && spriteImg.naturalWidth !== 0) { ctx.drawImage( spriteImg, player.x - 20, player.y - 22, 40, 40 ); } else { // Fallback circle if image is still loading ctx.fillStyle = '#2ecc71'; ctx.beginPath(); ctx.arc(player.x, player.y, player.radius, 0, Math.PI * 2); ctx.fill(); } if (player.isAttacking) { ctx.strokeStyle = '#f1c40f'; ctx.lineWidth = 4; ctx.beginPath(); ctx.arc(player.x, player.y, 35, 0, Math.PI * 2); ctx.stroke(); } }
